              onChange={(e) => setFilterStatus(e.target.value)}

              <option value="contacted">{lang.contacted}</option>
              <option value="negotiation">{lang.negotiation}</option>
              <option value="converted">{lang.converted}</option>
              <option value="notInterested">{lang.notInterested}</option>
            </select>
          </div>
        )}

        {/* Main Content */}
        {renderContent()}
      </div>

      {/* Lead Form Modal */}
      {showForm && (
        <LeadForm
          onSave={handleAddLead}
          onClose={() => {
            setShowForm(false);
            setEditingLead(null);
          }}
          initialData={editingLead}
          language={language}
          t={t}
        />
      )}

      {/* WhatsApp Modal */}
      {showNotificationModal && selectedLeadForNotif && (
        <NotificationModal
          lead={selectedLeadForNotif}
          type="whatsapp"
          templates={whatsappTemplates}
          onClose={() => setShowNotificationModal(false)}
          language={language}
          t={t}
        />
      )}

      {/* Email Modal */}
      {showEmailModal && selectedLeadForNotif && (
        <NotificationModal
          lead={selectedLeadForNotif}
          type="email"
          templates={emailTemplates}
          onClose={() => setShowEmailModal(false)}
          language={language}
          t={t}
        />
      )}
    </div>
  );
};

      email: '',
      phone: '',
      city: '',

      status: 'qualified',
      score: 70,
      notes: '',
      platform: 'facebook',
      assignedTo: 'Admin'

      ...prev,

const NotificationModal = ({ lead, type, templates, onClose, language, t }) => {
  const [selectedTemplate, setSelectedTemplate] = useState(templates[0]);
  const [customMessage, setCustomMessage] = useState(templates[0][type === 'whatsapp' ? 'message' : 'message']);

  const handleSend = () => {
    alert(`${type === 'whatsapp' ? 'WhatsApp' : 'Email'} sent to ${lead.name}!`);
    onClose();
  };

  const processMessage = (msg) => {
    return msg.replace(/{{name}}/g, lead.name);
  };

  return (
    <div className="fixed inset-0 bg-black/80 backdrop-blur-sm flex items-center justify-center z-50">
      <div className="bg-gradient-to-br from-green-900/40 to-black/40 border border-green-600/30 rounded-xl p-8 max-w-lg w-full mx-4">
        <h2 className="text-2xl font-bold text-white mb-6">
          {type === 'whatsapp' ? '💬 Send WhatsApp' : '📧 Send Email'} to {lead.name}
        </h2>

        <div className="space-y-4">
          <div>
            <label className="text-sm text-green-300/70 block mb-2">Select Template</label>
            <select
              value={selectedTemplate.id}
              onChange={(e) => {
                const template = templates.find(t => t.id === parseInt(e.target.value));
                setSelectedTemplate(template);
                setCustomMessage(template[type === 'whatsapp' ? 'message' : 'message']);
              }}
              className="w-full px-4 py-2 bg-black/40 border border-green-600/20 rounded-lg text-white focus:outline-none focus:border-green-500/60"
            >
              {templates.map(tmpl => (
                <option key={tmpl.id} value={tmpl.id}>{tmpl.name}</option>
              ))}
            </select>
          </div>

          <div>
            <label className="text-sm text-green-300/70 block mb-2">Message</label>
            <textarea
              value={customMessage}
              onChange={(e) => setCustomMessage(e.target.value)}
              rows="5"
              className="w-full px-4 py-2 bg-black/40 border border-green-600/20 rounded-lg text-white focus:outline-none focus:border-green-500/60"
            />
          </div>

          <div className="p-3 bg-green-900/20 border border-green-600/30 rounded-lg">
            <p className="text-xs text-green-300/70 mb-1">Preview:</p>
            <p className="text-sm text-green-200 whitespace-pre-wrap">
              {processMessage(customMessage)}
            </p>
          </div>

          <div className="flex gap-3 pt-4">
            <button
              onClick={handleSend}
              className="flex-1 px-4 py-2.5 bg-gradient-to-r from-green-600 to-green-700 hover:from-green-500 hover:to-green-600 text-white rounded-lg font-semibold transition-all"
            >
              Send
            </button>
            <button
              onClick={onClose}
              className="flex-1 px-4 py-2.5 bg-black/40 border border-green-600/20 text-green-300 rounded-lg font-semibold hover:bg-black/60 transition-colors"
            >
              Cancel
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};
